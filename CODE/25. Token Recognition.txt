
#include <stdio.h>
#include <ctype.h>
#include <string.h>

int main()
{
    char str[100];
    int i;

    printf("Enter an expression: ");
    scanf("%99s", str);

    printf("\nTokens:\n");

    for(i = 0; str[i] != '\0'; i++)
    {
        if(isalpha(str[i]))
            printf("%c - Identifier\n", str[i]);

        else if(isdigit(str[i]))
            printf("%c - Number\n", str[i]);

        else if(str[i] == '+' || str[i] == '-' ||
                str[i] == '*' || str[i] == '/')
            printf("%c - Operator\n", str[i]);

        else if(str[i] == '=')
            printf("%c - Assignment Operator\n", str[i]);
    }

    return 0;
}
