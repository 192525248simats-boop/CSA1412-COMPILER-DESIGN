
#include <stdio.h>
#include <string.h>

int main() {
    char a[20], b[20], A;

    printf("Enter non-terminal: ");
    scanf(" %c", &A);

    printf("Enter first production: ");
    scanf("%s", a);

    printf("Enter second production: ");
    scanf("%s", b);

    int i = 0;

    while (a[i] == b[i] && a[i] != '\0')
        i++;

    if (i == 0) {
        printf("No left factoring");
    } else {
        printf("%c -> ", A);
        for (int j = 0; j < i; j++)
            printf("%c", a[j]);

        printf("%c'\n", A);

        printf("%c' -> %s | %s\n", A, a+i, b+i);
    }

    return 0;
}
