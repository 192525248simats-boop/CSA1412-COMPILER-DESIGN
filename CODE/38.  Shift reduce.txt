#include <stdio.h>
#include <string.h>

int main() {
    char input[20], stack[20];
    int i = 0, top = -1;

    printf("Enter input: ");
    scanf("%s", input);

    strcat(input, "$");

    while(input[i] != '$') {
        stack[++top] = input[i++];
        stack[top + 1] = '\0';
        printf("Shift: %s\n", stack);

        if(stack[top] == 'd') {
            stack[top] = 'E';
            printf("Reduce: E -> id\n");
        }
    }

    if(top == 0 && stack[0] == 'E')
        printf("Accepted\n");
    else
        printf("Rejected\n");

    return 0;
}
