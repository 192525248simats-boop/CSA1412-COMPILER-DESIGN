#include <stdio.h>
#include <string.h>

int main() {
    char input[20];
    int i = 0;

    printf("Enter input string: ");
    scanf("%s", input);

    strcat(input, "$");

    printf("\nStack\tInput\tAction\n");

    while(input[i] != '$') {
        printf("%c\t%s\tShift\n", input[i], input + i);
        i++;
    }

    printf("$\t$\tAccept\n");

    return 0;
}
